# Функции меню
def get_main_markup():
    markup = types.InlineKeyboardMarkup(row_width=1)
    markup.add(
        types.InlineKeyboardButton("📢 Наш канал", url="https://t.me/playfunpay"),
        types.InlineKeyboardButton("💎 Темка дня", callback_data="get_temka"),
        types.InlineKeyboardButton("💸 Реклама", url="https://t.me/playfunpay1"),
        types.InlineKeyboardButton("❓ Вопросы и ответы", callback_data="faq")
    )
    return markup

def get_back_markup():
    markup = types.InlineKeyboardMarkup()
    markup.add(types.InlineKeyboardButton("⬅️ Назад", callback_data="back_to_main"))
    return markup

@bot.message_handler(commands=['start'])
def start(message):
    try:
        with open('IMG_20260320_161015_161.jpg', 'rb') as photo:
            bot.send_photo(message.chat.id, photo, caption=main_text, parse_mode='html', reply_markup=get_main_markup())
    except:
        bot.send_message(message.chat.id, main_text, parse_mode='html', reply_markup=get_main_markup())

@bot.callback_query_handler(func=lambda call: True)
def callback_inline(call):
    bot.answer_callback_query(call.id)
    user_id = call.from_user.id

    if call.data == "get_temka":
        step = user_progress.get(user_id, 0)
        if step < len(TEMKI):
            text = f"{TEMKI[step]}\n\n✅ <b>Завтра приходи за новой темкой!</b>"
            user_progress[user_id] = step + 1
        else:
            text = "🏆 <b>Ты изучил все 30 темок!</b>\n\nТеперь ты настоящий профи. Следи за каналом, скоро будет новый пак!"

        try:
            bot.edit_message_caption(text, call.message.chat.id, call.message.message_id, parse_mode='html', reply_markup=get_back_markup())
        except:
            bot.edit_message_text(text, call.message.chat.id, call.message.message_id, parse_mode='html', reply_markup=get_back_markup())

    elif call.data == "faq":
        faq_text = "<b>FAQ:</b>\n\n🔹 Как войти в команду? - Подпишись на канал.\n🔹 Куда писать вопрос? - Просто отправь его текстом в этот бот!"
        try:
            bot.edit_message_caption(faq_text, call.message.chat.id, call.message.message_id, parse_mode='html', reply_markup=get_back_markup())
        except:
            bot.edit_message_text(faq_text, call.message.chat.id, call.message.message_id, parse_mode='html', reply_markup=get_back_markup())

    elif call.data == "back_to_main":
        try:
            bot.edit_message_caption(main_text, call.message.chat.id, call.message.message_id, parse_mode='html', reply_markup=get_main_markup())
        except:
            bot.edit_message_text(main_text, call.message.chat.id, call.message.message_id, parse_mode='html', reply_markup=get_main_markup())
